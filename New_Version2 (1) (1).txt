/* =========================================================================
   Utilities: small helpers (escape, debounce, hex/base64)
   ========================================================================= */
function escapeHtml(s){ if(!s) return ''; return String(s).replace(/[&<>"'`=\/]/g, function(c){ return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;','/':'&#x2F;','`':'&#96;','=':'&#61;'}[c]; }); }
function debounce(fn, wait=300){ let t; return (...a)=>{ clearTimeout(t); t=setTimeout(()=>fn(...a), wait); }; }

/* =========================================================================
   Crypto: PBKDF2 hashing for password (Web Crypto)
   - store salt + derivedKey (hex)
   ========================================================================= */
async function generateSalt(len=16){
  const buf = crypto.getRandomValues(new Uint8Array(len));
  return btoa(String.fromCharCode(...buf));
}
async function deriveKey(password, salt, iterations=100_000){
  const enc = new TextEncoder();
  const passKey = await crypto.subtle.importKey('raw', enc.encode(password),'PBKDF2',false,['deriveBits']);
  const saltBytes = Uint8Array.from(atob(salt), c => c.charCodeAt(0));
  const bits = await crypto.subtle.deriveBits({name:'PBKDF2', salt: saltBytes, iterations, hash:'SHA-256'}, passKey, 256);
  const arr = new Uint8Array(bits);
  // return hex
  return Array.from(arr).map(b=>b.toString(16).padStart(2,'0')).join('');
}

/* =========================================================================
   IndexedDB wrapper for trades & media
   ========================================================================= */
const DB_NAME = 'qfx-db', DB_VER = 1;
function openDb(){ return new Promise((resolve, reject)=>{
  const req = indexedDB.open(DB_NAME, DB_VER);
  req.onupgradeneeded = e => {
    const db = e.target.result;
    if(!db.objectStoreNames.contains('trades')) db.createObjectStore('trades', { keyPath: 'id' });
    if(!db.objectStoreNames.contains('media')) db.createObjectStore('media', { keyPath: 'id' });
    if(!db.objectStoreNames.contains('users')) db.createObjectStore('users', { keyPath: 'username' });
  };
  req.onsuccess = () => resolve(req.result);
  req.onerror = () => reject(req.error);
}); }

async function dbPut(store, value){
  const db = await openDb();
  return new Promise((res, rej)=>{
    const tx = db.transaction(store, 'readwrite'), s = tx.objectStore(store);
    const r = s.put(value);
    r.onsuccess = ()=>res(r.result);
    r.onerror = ()=>rej(r.error);
  });
}
async function dbGetAll(store){
  const db = await openDb();
  return new Promise((res, rej)=>{
    const tx = db.transaction(store, 'readonly'), s = tx.objectStore(store);
    const r = s.getAll();
    r.onsuccess = ()=>res(r.result);
    r.onerror = ()=>rej(r.error);
  });
}
async function dbGet(store, key){
  const db = await openDb();
  return new Promise((res, rej)=>{
    const tx = db.transaction(store, 'readonly'), s = tx.objectStore(store);
    const r = s.get(key);
    r.onsuccess = ()=>res(r.result);
    r.onerror = ()=>rej(r.error);
  });
}
async function dbDelete(store, key){
  const db = await openDb();
  return new Promise((res, rej)=>{
    const tx = db.transaction(store, 'readwrite'), s = tx.objectStore(store);
    const r = s.delete(key);
    r.onsuccess = ()=>res();
    r.onerror = ()=>rej(r.error);
  });
}

/* =========================================================================
   Users: registration/login using IndexedDB users store (with derived key)
   ========================================================================= */
async function saveUserToDb(user){
  // user: { username, fullname, email, salt, derivedKey (hex), isVerified, ...}
  await dbPut('users', user);
}
async function getUserFromDb(usernameOrEmail){
  const all = await dbGetAll('users');
  return all.find(u=>u.username===usernameOrEmail || u.email===usernameOrEmail) || null;
}

/* =========================================================================
   Image helper: resize image to max width and return Blob
   ========================================================================= */
function resizeImageFile(file, maxWidth=1200, quality=0.8){
  return new Promise((res, rej)=>{
    if(!file.type.startsWith('image/')) return res(file);
    const img = new Image();
    const url = URL.createObjectURL(file);
    img.onload = ()=>{
      let {width, height} = img;
      if(width <= maxWidth){ URL.revokeObjectURL(url); return res(file); }
      const scale = maxWidth / width;
      const cv = document.createElement('canvas');
      cv.width = Math.round(width * scale);
      cv.height = Math.round(height * scale);
      const ctx = cv.getContext('2d');
      ctx.drawImage(img,0,0,cv.width,cv.height);
      cv.toBlob(blob=>{
        URL.revokeObjectURL(url);
        res(blob);
      }, 'image/jpeg', quality);
    };
    img.onerror = ()=>{ URL.revokeObjectURL(url); rej(new Error('Image load failed')); };
    img.src = url;
  });
}

/* Modal open/close with focus management */
let lastFocusedEl = null;
function openModal(html){
  lastFocusedEl = document.activeElement;
  const modal = document.getElementById('custom-modal');
  document.getElementById('modal-content').innerHTML = html;
  modal.classList.remove('hidden');
  document.body.classList.add('modal-open');
  // focus first focusable inside
  setTimeout(()=>{ const focusable = modal.querySelector('button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])'); if(focusable) focusable.focus(); }, 50);
  // Escape handler
  document.addEventListener('keydown', modalEscapeHandler);
}
function closeModal(){
  const modal = document.getElementById('custom-modal');
  modal.classList.add('hidden');
  document.body.classList.remove('modal-open');
  if(lastFocusedEl) lastFocusedEl.focus();
  document.removeEventListener('keydown', modalEscapeHandler);
}
function modalEscapeHandler(e){ if(e.key === 'Escape') closeModal(); }

/* =========================================================================
   Registration & Login flows
   ========================================================================= */
async function handleRegistration(e){
  e.preventDefault();
  const fullname = document.getElementById('reg-fullname').value.trim();
  const username = document.getElementById('reg-username').value.trim().toLowerCase();
  const email = document.getElementById('reg-email').value.trim().toLowerCase();
  const password = document.getElementById('reg-password').value;

  if(!fullname||!username||!email||!password){ showToast('Kosa','Tafadhali jaza taarifa zote','error'); return; }
  if(!checkPasswordStrength(password)){ showToast('Nenosiri','Nenosiri hafai', 'error'); return; }

  const existing = await getUserFromDb(username) || await getUserFromDb(email);
  if(existing){ showToast('Akaunti Ipo','Username au email imetumika', 'error'); return; }

  const salt = await generateSalt();
  const derivedKey = await deriveKey(password, salt);
  const user = { username, fullname, email, salt, derivedKey, isVerified: true }; // Keep isVerified true for demo / or implement OTP
  await saveUserToDb(user);
  showToast('Imefanikiwa','Usajili umefanikiwa. Ingia.', 'success');
  switchAuthTab('login');
  document.getElementById('login-identifier').value = username;
}

async function handleLogin(e){
  e.preventDefault();
  const identifier = document.getElementById('login-identifier').value.trim().toLowerCase();
  const password = document.getElementById('login-password').value;
  const errBox = document.getElementById('login-error');

  const user = await getUserFromDb(identifier);
  if(!user){ errBox.textContent='Hakuna akaunti'; errBox.classList.remove('hidden'); return; }

  // Verify password
  const derived = await deriveKey(password, user.salt);
  if(derived !== user.derivedKey){ errBox.textContent='Nenosiri si sahihi'; errBox.classList.remove('hidden'); return; }

  errBox.classList.add('hidden');
  APP.currentUser = user;
  // show app
  document.getElementById('auth-container').classList.add('hidden');
  document.getElementById('app-container').classList.remove('hidden');
  document.getElementById('hdr-name').textContent = user.fullname;
  // load data
  await loadUserDashboardData();
  switchView('dashboard');
  showToast('Karibu', `Habari ${user.fullname}`, 'success');
}

async function resetAccountAccess(username){
  const otp = document.getElementById('input-recovery-otp').value.trim();
  const newPwd = document.getElementById('input-new-password').value;
  const user = await dbGet('users', username);
  if(!user || user.recoveryCode !== otp){ showToast('Kosa','Msimbo si sahihi','error'); return; }
  if(!checkPasswordStrength(newPwd)){ showToast('Nenosiri','Nenosiri hafai', 'error'); return; }
  const salt = await generateSalt();
  const derivedKey = await deriveKey(newPwd, salt);
  user.salt = salt; user.derivedKey = derivedKey; user.recoveryCode = null;
  await saveUserToDb(user);
  closeModal();
  showToast('Imefanikiwa','Nenosiri limebadilishwa', 'success');
}

/* =========================================================================
   Trades: store trades in IndexedDB (media stored separately as Blobs)
   ========================================================================= */
async function saveMediaBlob(blob){
  // store in media store with generated id
  const id = 'm_' + Date.now() + '_' + Math.random().toString(36).slice(2,8);
  // keep type and blob as-is (IndexedDB supports blobs)
  await dbPut('media', { id, blob, createdAt: new Date().toISOString() });
  return id;
}
async function getMediaUrl(mediaId){
  const rec = await dbGet('media', mediaId);
  if(!rec) return null;
  // create object URL to use in <img> or <video>
  return URL.createObjectURL(rec.blob);
}
async function saveTradeToDb(trade){
  await dbPut('trades', trade);
}
async function getAllTrades(){
  const all = await dbGetAll('trades');
  // filter by current user
  return all.filter(t=>t.username === APP.currentUser.username).sort((a,b)=>b.date.localeCompare(a.date));
}

}

/* File select: resize image and store as blob in media store; for videos, store blob directly */
async function handleFileSelect(e, kind){
  const file = e.target.files[0];
  if(!file) return;
  if(file.type.startsWith('image/')){
    const resized = await resizeImageFile(file, 1200, 0.75);
    const id = await saveMediaBlob(resized);
    // keep temp reference in window for next save
    window['temp_' + kind] = id;
    showToast('Imechaguliwa', `${kind} imehifadhiwa kwa muda`, 'success');
  } else {
    // video
    const id = await saveMediaBlob(file);
    window['temp_' + kind] = id;
    showToast('Imechaguliwa', `${kind} video imehifadhiwa kwa muda`, 'success');
  }
}

/* Render Journal Table - uses safe text insertion (textContent) and creates buttons with handlers */
async function renderJournalTable(filter=''){
  const tbody = document.getElementById('journal-table-body');
  tbody.innerHTML = '';
  const trades = await getAllTrades();
  const filtered = trades.filter(t=>{
    if(!filter) return true;
    const q = filter.toUpperCase();
    return (t.pair||'').toUpperCase().includes(q) || (t.strategy||'').toUpperCase().includes(q) || (t.notes||'').toUpperCase().includes(q);
  });
  if(!filtered.length){
    const tr = document.createElement('tr');
    const td = document.createElement('td'); td.colSpan=7; td.className='p-4 text-gray-400'; td.textContent='Hakuna rekodi';
    tr.appendChild(td); tbody.appendChild(tr); return;
  }

  for(const t of filtered){
    const tr = document.createElement('tr'); tr.className='text-xs';
    const tdDate = document.createElement('td'); tdDate.textContent = t.date; tdDate.className='p-2';
    const tdPair = document.createElement('td'); tdPair.textContent = t.pair; tdPair.className='p-2 font-bold';
    const tdType = document.createElement('td'); tdType.textContent = t.type; tdType.className='p-2';
    const tdStrat = document.createElement('td'); tdStrat.textContent = t.strategy; tdStrat.className='p-2';
    const tdPnl = document.createElement('td'); tdPnl.textContent = (t.pnl>=0?'+':'') + '$' + Number(t.pnl).toFixed(2); tdPnl.className='p-2 font-mono ' + (t.pnl>=0?'text-emerald-400':'text-red-400');
    const tdMedia = document.createElement('td'); tdMedia.className='p-2';
    // media buttons
    if(t.mediaBefore){ const b = document.createElement('button'); b.className='px-2 py-1 bg-blue-600 rounded text-white text-xs'; b.textContent='Before'; b.onclick = async ()=>{ const url = await getMediaUrl(t.mediaBefore); openMediaViewer(url, 'Before - '+t.pair, 'image'); }; tdMedia.appendChild(b); }
    if(t.mediaAfter){ const b = document.createElement('button'); b.className='ml-2 px-2 py-1 bg-emerald-600 rounded text-white text-xs'; b.textContent='After'; b.onclick = async ()=>{ const url = await getMediaUrl(t.mediaAfter); openMediaViewer(url, 'After - '+t.pair, 'image'); }; tdMedia.appendChild(b); }
    if(t.mediaVideo){ const b = document.createElement('button'); b.className='ml-2 px-2 py-1 bg-purple-600 rounded text-white text-xs'; b.textContent='Video'; b.onclick = async ()=>{ const url = await getMediaUrl(t.mediaVideo); openMediaViewer(url, 'Video - '+t.pair, 'video'); }; tdMedia.appendChild(b); }

    tdAct.appendChild(del);

    tr.append(tdDate, tdPair, tdType, tdStrat, tdPnl, tdMedia, tdAct);
    tbody.appendChild(tr);
  }
}

/* Dashboard charts & stats */
async function renderDashboardChartsAndStats(){
  if(!APP.currentUser) return;
  const trades = await getAllTrades();
  let profits=0, losses=0, wins=0, lossesCount=0;
  trades.forEach(t=>{ if(t.pnl>=0){profits+=Number(t.pnl); wins++;} else {losses+=Math.abs(Number(t.pnl)); lossesCount++;} });
  const net = profits - losses; const total=trades.length; const winRate = total?((wins/total)*100).toFixed(1):'0.0';
  document.getElementById('stat-net-pnl').textContent = (net>=0?'+':'-') + '$' + Math.abs(net).toFixed(2);
  document.getElementById('stat-profit').textContent = '$' + profits.toFixed(2);
  document.getElementById('stat-loss').textContent = '$' + losses.toFixed(2);
  document.getElementById('stat-winrate').textContent = winRate + '%';

  // simple charts (if Chart present)
  if(window.Chart){
    const sorted = [...trades].sort((a,b)=>new Date(a.date)-new Date(b.date));
    const labels=['Start']; const data=[0]; let run=0;
    sorted.forEach(t=>{ run += Number(t.pnl); labels.push(t.date); data.push(run); });
    // equity
    const eqCtx = document.getElementById('equityChart').getContext('2d');
    if(window._eqChart) window._eqChart.destroy();
    window._eqChart = new Chart(eqCtx, { type:'line', data:{ labels, datasets:[{ label:'Cumulative', data, borderColor:'#3b82f6', backgroundColor:'rgba(59,130,246,0.12)', fill:true }]}, options:{responsive:true, maintainAspectRatio:false}});
    // win/loss donut
    const wlCtx = document.getElementById('winLossChart').getContext('2d');
    if(window._wlChart) window._wlChart.destroy();
    window._wlChart = new Chart(wlCtx, { type:'doughnut', data:{ labels:['Wins','Losses'], datasets:[{ data:[wins, lossesCount], backgroundColor:['#10b981','#ef4444'] }]}, options:{responsive:true, maintainAspectRatio:false}});
  }
}

function changeCalendarMonth(delta){ APP.currentCalendar = APP.currentCalendar || new Date(); APP.currentCalendar.setMonth(APP.currentCalendar.getMonth()+delta); renderCalendarView(); }
async function renderCalendarView(){
  if(!APP.currentUser) return;
  const now = APP.currentCalendar || new Date();
  const year = now.getFullYear(), month = now.getMonth();
  const monthNames = ["Januari","Februari","Machi","Aprili","Mei","Juni","Julai","Agosti","Septemba","Oktoba","Novemba","Desemba"];
  document.getElementById('calendar-month-title').textContent = monthNames[month] + ' ' + year;
  const firstDay = new Date(year, month, 1).getDay();
  const daysInMonth = new Date(year, month+1, 0).getDate();
  const grid = document.getElementById('calendar-grid'); grid.innerHTML='';
  // collect P&L per date
  const trades = await getAllTrades(); const pnlByDate = {};
  trades.forEach(t => { pnlByDate[t.date] = (pnlByDate[t.date] || 0) + Number(t.pnl); });

  for(let i=0;i<firstDay;i++) grid.innerHTML += '<div class="h-20 rounded-xl bg-slate-900/30 border"></div>';
  for(let d=1; d<=daysInMonth; d++){
    const date = new Date(year, month, d);
    const dateStr = date.toISOString().split('T')[0];
    const pnl = pnlByDate[dateStr] || 0;
    const color = pnl>0 ? 'text-emerald-400' : pnl<0 ? 'text-red-400' : 'text-gray-400';
    grid.innerHTML += `<div class="h-20 rounded-xl bg-slate-900/70 p-2 flex flex-col justify-between"><div class="flex justify-between text-xs"><span>${d}</span><span class="${color}">${pnl?((pnl>0?'+':'-')+'$'+Math.abs(pnl).toFixed(2)) : ''}</span></div></div>`;
  }
}

/* Export CSV */
async function exportCSV(){
  const trades = await getAllTrades();
  if(!trades.length){ showToast('Hakuna','Hakuna trades za ku-export','info'); return; }
  const rows = [['date','pair','type','strategy','pnl','notes']];
  trades.forEach(t=> rows.push([t.date,t.pair,t.type,t.strategy,t.pnl, (t.notes||'').replace(/\n/g,' ')]));
  const csv = rows.map(r=>r.map(c=>`"${String(c).replace(/"/g,'""')}"`).join(',')).join('\n');
  const blob = new Blob([csv], { type:'text/csv' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a'); a.href=url; a.download = 'trades.csv'; a.click(); URL.revokeObjectURL(url);
}

function checkPasswordStrength(password){
  const len = document.getElementById('pwd-req-len'), mix = document.getElementById('pwd-req-mix');
  const isLen = password.length>=8;
  const isMix = /[A-Z]/.test(password)&&/[a-z]/.test(password)&&/[0-9]/.test(password)&&/[^A-Za-z0-9]/.test(password);
  len.className = isLen? 'text-emerald-400':'text-red-400'; mix.className = isMix? 'text-emerald-400':'text-red-400';
